cmake_minimum_required(VERSION 3.22)
project(katevoicepanel VERSION 0.1.0 LANGUAGES CXX)

find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
include(KDECMakeSettings)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets WebEngineWidgets WebChannel)
find_package(KF6 REQUIRED COMPONENTS CoreAddons TextEditor XmlGui I18n)

kcoreaddons_add_plugin(katevoicepanel INSTALL_NAMESPACE "kf6/ktexteditor")
target_sources(katevoicepanel PRIVATE src/plugin.cpp src/panel.cpp src/filelink.cpp src/voiceactivity.cpp)
target_link_libraries(katevoicepanel PRIVATE KF6::TextEditor KF6::CoreAddons KF6::XmlGui KF6::I18n Qt6::WebEngineWidgets Qt6::WebChannel)
target_compile_definitions(katevoicepanel PRIVATE TRANSLATION_DOMAIN="katevoicepanel")
file(GLOB_RECURSE PANEL_ASSETS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/web/*")
set(BRIDGE_ASSETS bridge/main.mjs bridge/deepseek.mjs bridge/dsh.mjs bridge/profile.mjs bridge/readonly.mjs bridge/tools.mjs bridge/audio.mjs)
qt_add_resources(katevoicepanel voice_assets PREFIX "/voicepanel" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${PANEL_ASSETS} ${BRIDGE_ASSETS} ui.rc)

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
    find_program(NODE_EXECUTABLE node REQUIRED)
    add_test(NAME bridge COMMAND ${NODE_EXECUTABLE} --test ${CMAKE_CURRENT_SOURCE_DIR}/tests/bridge_test.mjs)
    add_test(NAME renderer COMMAND ${NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/renderer_test.cjs)
    add_executable(filelink_test tests/filelink_test.cpp src/filelink.cpp)
    target_include_directories(filelink_test PRIVATE src)
    target_link_libraries(filelink_test PRIVATE Qt6::Core Qt6::Test)
    add_test(NAME filelink COMMAND filelink_test)
    add_executable(panel_test tests/panel_test.cpp src/panel.cpp src/filelink.cpp src/voiceactivity.cpp)
    target_include_directories(panel_test PRIVATE src)
    target_link_libraries(panel_test PRIVATE KF6::TextEditor KF6::I18n KF6::CoreAddons KF6::XmlGui Qt6::WebEngineWidgets Qt6::WebChannel Qt6::Test)
    target_compile_definitions(panel_test PRIVATE TRANSLATION_DOMAIN="katevoicepanel" SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" PLUGIN_FILE="$<TARGET_FILE:katevoicepanel>")
    add_dependencies(panel_test katevoicepanel)
    qt_add_resources(panel_test test_assets PREFIX "/voicepanel" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${PANEL_ASSETS} ${BRIDGE_ASSETS})
    add_test(NAME panel COMMAND panel_test)
    add_executable(voiceactivity_test tests/voiceactivity_test.cpp src/voiceactivity.cpp)
    target_include_directories(voiceactivity_test PRIVATE src)
    target_link_libraries(voiceactivity_test PRIVATE Qt6::Core Qt6::Test)
    add_test(NAME voiceactivity COMMAND voiceactivity_test)
    set_tests_properties(bridge PROPERTIES TIMEOUT 90 LABELS "protocol;offline")
    set_tests_properties(filelink renderer voiceactivity PROPERTIES TIMEOUT 30 LABELS "rendering;offline")
    set_tests_properties(panel PROPERTIES TIMEOUT 120 LABELS "integration;offline")
endif()
